## league

Task: Five teams play in a league, listed in this order: Lions, Tigers, Bears, Wolves, Hawks. The results, as home team, home goals, away team, away goals: Lions 2 Tigers 1; Bears 0 Wolves 0; Hawks 3 Lions 3; Lions 1 Bears 3; Tigers 2 Wolves 2; Hawks 1 Bears 1; Wolves 1 Lions 0; Tigers 4 Hawks 1; Bears 1 Tigers 2; Wolves 2 Hawks 4. A win is worth 3 points, a draw 1, a loss 0. Print the standings, one team per line as its name, points and goal difference (goals scored minus goals conceded) separated by single spaces, like `Lions 4 -2`. Order: most points first; on equal points, the larger goal difference first; if that is equal too, more goals scored first.

### attempt 1: OK

```cl
fn main() uses io
  let names = ["Lions", "Tigers", "Bears", "Wolves", "Hawks"]
  let hs = [0, 2, 4, 0, 1, 4, 3, 1, 2, 3]
  let hg = [2, 0, 3, 1, 2, 1, 1, 4, 1, 2]
  let aw = [1, 3, 0, 2, 3, 2, 0, 4, 1, 4]
  let ag = [1, 0, 3, 3, 2, 1, 0, 1, 2, 4]
  var pts = repeat(item: 0, times: 5)
  var gf = repeat(item: 0, times: 5)
  var gc = repeat(item: 0, times: 5)
  for m in 0 until 10
    let h = hs[m]
    let a = aw[m]
    let x = hg[m]
    let y = ag[m]
    gf[h] = gf[h] + x
    gc[h] = gc[h] + y
    gf[a] = gf[a] + y
    gc[a] = gc[a] + x
    if x > y
      pts[h] = pts[h] + 3
    else if x == y
      pts[h] = pts[h] + 1
      pts[a] = pts[a] + 1
    else
      pts[a] = pts[a] + 3
    end if
  end for
  var keys = repeat(item: 0, times: 5)
  for t in 0 until 5
    keys[t] = pts[t] * 10000 + (gf[t] - gc[t] + 100) * 100 + gf[t]
  end for
  var order = [0, 1, 2, 3, 4]
  for p in 0 until 5
    for j in 0 until 4
      if keys[order[j]] < keys[order[j + 1]]
        let tmp = order[j]
        order[j] = order[j + 1]
        order[j + 1] = tmp
      end if
    end for
  end for
  for r in 0 until 5
    let k = order[r]
    let gd = gf[k] - gc[k]
    print("{names[k]} {pts[k]} {gd}")
  end for
end fn main
```

## orders

Task: A shop stocks three items, each with a price and a quantity in stock: apple (price 3, 10 in stock), bread (price 5, 2 in stock), cheese (price 12, 1 in stock). Process these orders in turn, each a customer, an item and a quantity: Ann apple 4; Bob bread 3; Cy cheese 1; Ann bread 2; Dee cheese 1; Bob apple 6; Cy fig 1; Ann apple 1. For each order print one line: if the shop doesn't sell the item, `<customer> <item> unknown`; if there isn't enough in stock, `<customer> <item> short` (nothing is sold); otherwise the order is filled, the stock goes down, and the line is `<customer> <item> <quantity> <cost>`, where cost is quantity times price. After all the orders, print each item's remaining stock as `<item> <quantity>` in the order listed above, then each customer's total spending as `<customer> <total>` in the order they first ordered (including customers who bought nothing), then the shop's total revenue as a bare number.

### attempt 1: OK

```cl
fn indexOf(xs: List[String], x: String) -> Int
  example indexOf(xs: ["a", "b"], x: "b") == 1
  example indexOf(xs: [], x: "a") == -1
  for i in 0 until length(xs)
    if xs[i] == x
      return i
    end if
  end for
  return -1
end fn indexOf

fn main() uses io
  let names: List[String] = ["apple", "bread", "cheese"]
  let prices: List[Int] = [3, 5, 12]
  var stock: List[Int] = [10, 2, 1]
  let custs: List[String] = ["Ann", "Bob", "Cy", "Ann", "Dee", "Bob", "Cy", "Ann"]
  let items: List[String] = ["apple", "bread", "cheese", "bread", "cheese", "apple", "fig", "apple"]
  let qtys: List[Int] = [4, 3, 1, 2, 1, 6, 1, 1]
  var cnames: List[String] = []
  var spent: List[Int] = []
  var revenue = 0

  for who in custs
    if indexOf(xs: cnames, x: who) < 0
      cnames = append(list: cnames, item: who)
      spent = append(list: spent, item: 0)
    end if
  end for

  for i in 0 until length(custs)
    let cust = custs[i]
    let item = items[i]
    let q = qtys[i]
    let ci = indexOf(xs: cnames, x: cust)
    let ni = indexOf(xs: names, x: item)
    if ni < 0
      print("{cust} {item} unknown")
    else if stock[ni] < q
      print("{cust} {item} short")
    else
      stock[ni] = stock[ni] - q
      let cost = q * prices[ni]
      spent[ci] = spent[ci] + cost
      revenue = revenue + cost
      print("{cust} {item} {q} {cost}")
    end if
  end for

  for j in 0 until length(names)
    print("{names[j]} {stock[j]}")
  end for

  for k in 0 until length(cnames)
    print("{cnames[k]} {spent[k]}")
  end for

  print(revenue)
end fn main
```

## life

Task: Run Conway's Game of Life on a grid of 8 rows and 10 columns. Cells beyond the edges count as dead (the grid does not wrap around). Rows and columns are numbered from 0; the live cells at the start, as (row, column), are (0, 1), (1, 2), (2, 0), (2, 1), (2, 2), (5, 9), (6, 9), (7, 9). A live cell with 2 or 3 live neighbours (of its 8) stays alive, a dead cell with exactly 3 becomes alive, and every other cell is dead in the next generation. Run 6 generations. First print one line with the number of live cells after each generation, separated by single spaces. Then print the grid after the last generation, one line per row, `#` for live and `.` for dead.

### attempt 1: OK

```cl
fn neighbors(g: List[List[Int]], r: Int, c: Int) -> Int
  var count = 0
  for dr in -1 through 1
    for dc in -1 through 1
      if not (dr == 0 and dc == 0)
        let nr = r + dr
        let nc = c + dc
        if nr >= 0 and nr < 8 and nc >= 0 and nc < 10
          count = count + g[nr][nc]
        end if
      end if
    end for
  end for
  return count
end fn neighbors

fn step(g: List[List[Int]]) -> List[List[Int]]
  var ng = repeat(item: repeat(item: 0, times: 10), times: 8)
  for r in 0 until 8
    for c in 0 until 10
      let n = neighbors(g: g, r: r, c: c)
      let cur = g[r][c]
      if cur == 1 and (n == 2 or n == 3)
        ng[r][c] = 1
      else if cur == 0 and n == 3
        ng[r][c] = 1
      end if
    end for
  end for
  return ng
end fn step

fn countLive(g: List[List[Int]]) -> Int
  var total = 0
  for row in g
    for x in row
      total = total + x
    end for
  end for
  return total
end fn countLive

fn main() uses io
  var g = repeat(item: repeat(item: 0, times: 10), times: 8)
  g[0][1] = 1
  g[1][2] = 1
  g[2][0] = 1
  g[2][1] = 1
  g[2][2] = 1
  g[5][9] = 1
  g[6][9] = 1
  g[7][9] = 1
  var line = ""
  for gen in 1 through 6
    g = step(g)
    let n = countLive(g)
    if gen == 1
      line = "{n}"
    else
      line = "{line} {n}"
    end if
  end for
  print(line)
  for i in 0 until 8
    var s = ""
    for j in 0 until 10
      if g[i][j] == 1
        s = "{s}#"
      else
        s = "{s}."
      end if
    end for
    print(s)
  end for
end fn main
```

## adventure

Task: Write a small text adventure that runs a fixed list of commands. Rooms: Hall, Library, Kitchen, Vault. Exits: from Hall, north goes to Library and east to Kitchen; from Library, south goes to Hall; from Kitchen, west goes to Hall and down to Vault; from Vault, up goes to Kitchen. The way down from Kitchen is locked unless the player carries the key. Items at the start: Library has key, Kitchen has bread, Vault has gold. The player starts in Hall carrying nothing. Commands: `go <direction>` moves and prints the new room's name, or prints `You can't go <direction>.` if there's no such exit, or `The way down is locked.` if it's the locked way and the player has no key. `take <item>` moves the item from the room to the player and prints `Taken: <item>.`, or prints `There is no <item> here.`. `look` prints the room's name, then a line with the room's items separated by single spaces, or `nothing`. `inventory` prints the carried items in the order they were taken, separated by single spaces, or `nothing`. Run these commands in order: go east, take key, take bread, go down, go west, go north, look, take key, go south, go east, go down, take gold, look, go up, go up, inventory.

### attempt 1: OK

```cl
enum Room
  Hall
  Library
  Kitchen
  Vault
end enum Room

enum Dir
  North
  South
  East
  West
  Down
  Up
end enum Dir

enum Cmd
  Go(dir: Dir)
  Take(item: String)
  Look
  Inventory
end enum Cmd

struct World
  room: Room
  hall: List[String]
  library: List[String]
  kitchen: List[String]
  vault: List[String]
  inv: List[String]
end struct World

fn roomName(r: Room) -> String
  var s = ""
  match r
    case Hall
      s = "Hall"
    case Library
      s = "Library"
    case Kitchen
      s = "Kitchen"
    case Vault
      s = "Vault"
  end match
  return s
end fn roomName

fn dirName(d: Dir) -> String
  var s = ""
  match d
    case North
      s = "north"
    case South
      s = "south"
    case East
      s = "east"
    case West
      s = "west"
    case Down
      s = "down"
    case Up
      s = "up"
  end match
  return s
end fn dirName

fn move(r: Room, d: Dir) -> Option[Room]
  var result: Option[Room] = None
  match r
    case Hall
      if d == North
        result = Some(Library)
      else if d == East
        result = Some(Kitchen)
      end if
    case Library
      if d == South
        result = Some(Hall)
      end if
    case Kitchen
      if d == West
        result = Some(Hall)
      else if d == Down
        result = Some(Vault)
      end if
    case Vault
      if d == Up
        result = Some(Kitchen)
      end if
  end match
  return result
end fn move

fn contains(xs: List[String], x: String) -> Bool
  example contains(xs: ["a", "b"], x: "b") == true
  example contains(xs: [], x: "b") == false
  for y in xs
    if y == x
      return true
    end if
  end for
  return false
end fn contains

fn remove(xs: List[String], x: String) -> List[String]
  example remove(xs: ["a", "b", "c"], x: "b") == ["a", "c"]
  var out: List[String] = []
  var done = false
  for y in xs
    if not done and y == x
      done = true
    else
      out = append(list: out, item: y)
    end if
  end for
  return out
end fn remove

fn joinWords(xs: List[String]) -> String
  example joinWords([]) == "nothing"
  example joinWords(["a", "b"]) == "a b"
  if length(xs) == 0
    return "nothing"
  end if
  var line = ""
  for i in 0 until length(xs)
    if i == 0
      line = xs[i]
    else
      line = "{line} {xs[i]}"
    end if
  end for
  return line
end fn joinWords

fn itemsOf(w: World, r: Room) -> List[String]
  var s: List[String] = []
  match r
    case Hall
      s = w.hall
    case Library
      s = w.library
    case Kitchen
      s = w.kitchen
    case Vault
      s = w.vault
  end match
  return s
end fn itemsOf

fn setItems(w: World, r: Room, xs: List[String]) -> World
  var n = w
  match r
    case Hall
      n.hall = xs
    case Library
      n.library = xs
    case Kitchen
      n.kitchen = xs
    case Vault
      n.vault = xs
  end match
  return n
end fn setItems

fn step(w: World, c: Cmd) -> World uses io
  var w2 = w
  match c
    case Go(d)
      match move(r: w.room, d: d)
        case Some(nr)
          if w.room == Kitchen and d == Down and not contains(xs: w.inv, x: "key")
            print("The way down is locked.")
          else
            w2.room = nr
            print(roomName(nr))
          end if
        case None
          print("You can't go {dirName(d)}.")
      end match
    case Take(item)
      let here = itemsOf(w: w, r: w.room)
      if contains(xs: here, x: item)
        w2 = setItems(w: w2, r: w.room, xs: remove(xs: here, x: item))
        w2.inv = append(list: w2.inv, item: item)
        print("Taken: {item}.")
      else
        print("There is no {item} here.")
      end if
    case Look
      print(roomName(w.room))
      print(joinWords(itemsOf(w: w, r: w.room)))
    case Inventory
      print(joinWords(w.inv))
  end match
  return w2
end fn step

fn main() uses io
  let cmds: List[Cmd] = [Go(East), Take("key"), Take("bread"), Go(Down), Go(West), Go(North), Look, Take("key"), Go(South), Go(East), Go(Down), Take("gold"), Look, Go(Up), Go(Up), Inventory]
  var w = World(room: Hall, hall: [], library: ["key"], kitchen: ["bread"], vault: ["gold"], inv: [])
  for c in cmds
    w = step(w: w, c: c)
  end for
end fn main
```
