## sieve

Task: Use the sieve of Eratosthenes (a list of Bools) to find the primes below 100. Print how many there are, then the largest one.

### attempt 1: OK

```cl
fn main() uses io
  var sieve = repeat(item: true, times: 100)
  sieve[0] = false
  sieve[1] = false
  var i = 2
  while i * i < 100
    if sieve[i]
      var j = i * i
      while j < 100
        sieve[j] = false
        j = j + i
      end while
    end if
    i = i + 1
  end while
  var count = 0
  var largest = 0
  var k = 0
  while k < 100
    if sieve[k]
      count = count + 1
      largest = k
    end if
    k = k + 1
  end while
  print(count)
  print(largest)
end fn main
```

## sort

Task: Sort [5, 2, 8, 1, 9, 3] with a sorting algorithm you write yourself and print the sorted list.

### attempt 1: OK

```cl
fn sortList(xs: List[Int]) -> List[Int]
  example sortList([3, 1, 2]) == [1, 2, 3]
  example sortList([1, 2, 3]) == [1, 2, 3]
  var ys = xs
  var i = 1
  while i < length(ys)
    let key = ys[i]
    var j = i - 1
    while j >= 0 and ys[j] > key
      ys[j + 1] = ys[j]
      j = j - 1
    end while
    ys[j + 1] = key
    i = i + 1
  end while
  return ys
end fn sortList

fn main() uses io
  print(sortList([5, 2, 8, 1, 9, 3]))
end fn main
```
