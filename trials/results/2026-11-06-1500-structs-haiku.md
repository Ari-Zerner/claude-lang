## point-move

Task: Define a point type with x and y. Write a function that returns a point moved by dx and dy. Move the point (2, 3) by (1, -1) and print the new point's x, then its y, on separate lines.

### attempt 1: OK

```cl
struct Point
  x: Int
  y: Int
end struct Point

fn move(p: Point, dx: Int, dy: Int) -> Point
  example move(p: Point(x: 2, y: 3), dx: 1, dy: -1) == Point(x: 3, y: 2)
  example move(p: Point(x: 0, y: 0), dx: 0, dy: 0) == Point(x: 0, y: 0)
  return Point(x: p.x + dx, y: p.y + dy)
end fn move

fn main() uses io
  let start = Point(x: 2, y: 3)
  let moved = move(p: start, dx: 1, dy: -1)
  print(moved.x)
  print(moved.y)
end fn main
```

## segment

Task: A line segment has a start point and an end point. Write a function that returns a segment's squared length. Make the segment from (3, 4) to (6, 8) and print its squared length; then change its start point's x to 0 and print its squared length again.

### attempt 1: OK

```cl
struct Point
  x: Int
  y: Int
end struct Point

struct Segment
  start: Point
  finish: Point
end struct Segment

fn sq_len(seg: Segment) -> Int
  example sq_len(Segment(start: Point(x: 3, y: 4), finish: Point(x: 6, y: 8))) == 25
  example sq_len(Segment(start: Point(x: 0, y: 4), finish: Point(x: 6, y: 8))) == 52
  let dx = seg.finish.x - seg.start.x
  let dy = seg.finish.y - seg.start.y
  return dx * dx + dy * dy
end fn sq_len

fn main() uses io
  var seg = Segment(start: Point(x: 3, y: 4), finish: Point(x: 6, y: 8))
  print(sq_len(seg))
  var s = seg.start
  s.x = 0
  seg.start = s
  print(sq_len(seg))
end fn main
```

## tree

Task: Define a tree where each node has an Int value and any number of children. Build the tree whose root is 1, with children 2 (whose children are 4 and 5) and 3. Print the sum of all values, then the tree's depth (a single node has depth 1).

### attempt 1: OK

```cl
struct Tree
  value: Int
  children: List[Tree]
end struct Tree

fn sum_tree(t: Tree) -> Int
  var total = t.value
  for c in t.children
    total = total + sum_tree(c)
  end for
  return total
end fn sum_tree

fn depth(t: Tree) -> Int
  var best = 0
  for c in t.children
    let d = depth(c)
    if d > best
      best = d
    end if
  end for
  return 1 + best
end fn depth

fn main() uses io
  let empty: List[Tree] = []
  let n4 = Tree(value: 4, children: empty)
  let n5 = Tree(value: 5, children: empty)
  let n2 = Tree(value: 2, children: [n4, n5])
  let n3 = Tree(value: 3, children: empty)
  let root = Tree(value: 1, children: [n2, n3])
  print(sum_tree(root))
  print(depth(root))
end fn main
```

## by-age

Task: People have a name and an age: Ann is 31, Bob is 25, Cy is 40. Sort them by age, youngest first, with a sort you write yourself, and print their names, one per line.

### attempt 1: OK

```cl
struct Person
  name: String
  age: Int
end struct Person

fn insert_sorted(sofar: List[Person], p: Person) -> List[Person]
  var out: List[Person] = []
  var done = false
  for q in sofar
    if not done and p.age < q.age
      out = append(list: out, item: p)
      done = true
    end if
    out = append(list: out, item: q)
  end for
  if not done
    out = append(list: out, item: p)
  end if
  return out
end fn insert_sorted

fn sorted_by_age(people: List[Person]) -> List[Person]
  example sorted_by_age(people: [Person(name: "Cy", age: 40), Person(name: "Ann", age: 31)]) == [Person(name: "Ann", age: 31), Person(name: "Cy", age: 40)]
  example sorted_by_age(people: []) == []
  var result: List[Person] = []
  for p in people
    result = insert_sorted(sofar: result, p: p)
  end for
  return result
end fn sorted_by_age

fn main() uses io
  var people: List[Person] = []
  people = append(list: people, item: Person(name: "Ann", age: 31))
  people = append(list: people, item: Person(name: "Bob", age: 25))
  people = append(list: people, item: Person(name: "Cy", age: 40))
  var ordered = sorted_by_age(people: people)
  for p in ordered
    print(p.name)
  end for
end fn main
```
