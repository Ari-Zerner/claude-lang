## fib-list

Task: Build a list of the first 10 Fibonacci numbers (starting 0, 1) and print the list.

### attempt 1: OK

```cl
fn main() uses io
  var fibs: List[Int] = []
  var a = 0
  var b = 1
  for i in 0 until 10
```

## sieve

Task: Use the sieve of Eratosthenes (a list of Bools) to find the primes below 100. Print how many there are, then the largest one.

### attempt 1: OK

```cl
fn main() uses io
  let n = 100
  var sieve = repeat(item: true, times: n)
  sieve[0] = false
  sieve[1] = false
  var count = 0
  var largest = 0
  for i in 2 until n
    if sieve[i]
      count = count + 1
      largest = i
      var j = i * i
      while j < n
        sieve[j] = false
        j = j + i
      end while
    end if
  end for
  print(count)
  print(largest)
end fn main
```

## dedupe

Task: Remove duplicates from [3, 1, 3, 2, 1, 5], keeping the first occurrence of each, and print the resulting list.

### attempt 1: OK

```cl
fn contains(xs: List[Int], x: Int) -> Bool
  example contains(xs: [1, 2, 3], x: 2) == true
  example contains(xs: [1, 2, 3], x: 5) == false
  for y in xs
    if y == x
      return true
    end if
  end for
  return false
end fn contains

fn dedupe(xs: List[Int]) -> List[Int]
  example dedupe([3, 1, 3, 2, 1, 5]) == [3, 1, 2, 5]
  example dedupe([]) == []
  var result: List[Int] = []
  for x in xs
    if not contains(xs: result, x: x)
      result = append(list: result, item: x)
    end if
  end for
  return result
end fn dedupe

fn main() uses io
  print(dedupe([3, 1, 3, 2, 1, 5]))
end fn main
```

## words

Task: Given the list of words apple, kiwi, banana, fig: print the longest word, then the total number of characters in all the words.

### attempt 1: OK

```cl
fn longest(words: List[String]) -> String
  example longest(["apple", "kiwi", "banana", "fig"]) == "banana"
  var best = words[0]
  for w in words
    if length(w) > length(best)
      best = w
    end if
  end for
  return best
end fn longest

fn totalChars(words: List[String]) -> Int
  example totalChars(["apple", "kiwi", "banana", "fig"]) == 18
  var sum = 0
  for w in words
    sum = sum + length(w)
  end for
  return sum
end fn totalChars

fn main() uses io
  let words = ["apple", "kiwi", "banana", "fig"]
  print(longest(words))
  print(totalChars(words))
end fn main
```

## sort

Task: Sort [5, 2, 8, 1, 9, 3] with a sorting algorithm you write yourself and print the sorted list.

### attempt 1: OK

```cl
fn insertion_sort(xs: List[Int]) -> List[Int]
  example insertion_sort([3, 1, 2]) == [1, 2, 3]
  example insertion_sort([]) == []
  var arr = xs
  for i in 1 until length(arr)
    let key = arr[i]
    var j = i - 1
    while j >= 0 and arr[j] > key
      arr[j + 1] = arr[j]
      j = j - 1
    end while
    arr[j + 1] = key
  end for
  return arr
end fn insertion_sort

fn main() uses io
  let xs = [5, 2, 8, 1, 9, 3]
  print(insertion_sort(xs))
end fn main
```

## merge

Task: Write a function that merges two sorted lists into one sorted list. Print the merge of [1, 4, 9] and [2, 3, 10, 11].

### attempt 1: OK

```cl
fn merge(a: List[Int], b: List[Int]) -> List[Int]
  example merge(a: [1, 4, 9], b: [2, 3, 10, 11]) == [1, 2, 3, 4, 9, 10, 11]
  example merge(a: [], b: [1]) == [1]
  example merge(a: [], b: []) == []
  var result: List[Int] = []
  var i = 0
  var j = 0
  while i < length(a) and j < length(b)
    if a[i] <= b[j]
      result = append(list: result, item: a[i])
      i = i + 1
    else
      result = append(list: result, item: b[j])
      j = j + 1
    end if
  end while
  while i < length(a)
    result = append(list: result, item: a[i])
    i = i + 1
  end while
  while j < length(b)
    result = append(list: result, item: b[j])
    j = j + 1
  end while
  return result
end fn merge

fn main() uses io
  print(merge(a: [1, 4, 9], b: [2, 3, 10, 11]))
end fn main
```
