## sieve

Task: Use the sieve of Eratosthenes (a list of Bools) to find the primes below 100. Print how many there are, then the largest one.

### attempt 1: OK

```cl
fn sieve(limit: Int) -> List[Bool]
  example sieve(limit: 5) == [false, false, true, true, false]
  var flags = repeat(item: true, times: limit)
  if limit > 0
    flags[0] = false
  end if
  if limit > 1
    flags[1] = false
  end if
  var p = 2
  while p * p < limit
    if flags[p]
      var m = p * p
      while m < limit
        flags[m] = false
        m = m + p
      end while
    end if
    p = p + 1
  end while
  return flags
end fn sieve

fn main() uses io
  let flags = sieve(limit: 100)
  var count = 0
  var largest = 0
  for i in 0 until 100
    if flags[i]
      count = count + 1
      largest = i
    end if
  end for
  print(count)
  print(largest)
end fn main
```

## dedupe

Task: Remove duplicates from [3, 1, 3, 2, 1, 5], keeping the first occurrence of each, and print the resulting list.

### attempt 1: OK

```cl
fn contains(xs: List[Int], x: Int) -> Bool
  example contains(xs: [1, 2, 3], x: 2) == true
  example contains(xs: [], x: 2) == false
  for y in xs
    if y == x
      return true
    end if
  end for
  return false
end fn contains

fn dedupe(xs: List[Int]) -> List[Int]
  example dedupe(xs: [3, 1, 3, 2, 1, 5]) == [3, 1, 2, 5]
  var out: List[Int] = []
  for x in xs
    if not contains(xs: out, x: x)
      out = append(list: out, item: x)
    end if
  end for
  return out
end fn dedupe

fn main() uses io
  print(dedupe(xs: [3, 1, 3, 2, 1, 5]))
end fn main
```

## words

Task: Given the list of words apple, kiwi, banana, fig: print the longest word, then the total number of characters in all the words.

### attempt 1: OK

```cl
fn longest(words: List[String]) -> String
  example longest(words: ["apple", "kiwi"]) == "apple"
  var best = ""
  for w in words
    if length(w) > length(best)
      best = w
    end if
  end for
  return best
end fn longest

fn total_chars(words: List[String]) -> Int
  example total_chars(words: ["apple", "kiwi"]) == 9
  var sum = 0
  for w in words
    sum = sum + length(w)
  end for
  return sum
end fn total_chars

fn main() uses io
  let words: List[String] = ["apple", "kiwi", "banana", "fig"]
  print("{longest(words: words)}")
  print("{total_chars(words: words)}")
end fn main
```

## sort

Task: Sort [5, 2, 8, 1, 9, 3] with a sorting algorithm you write yourself and print the sorted list.

### attempt 1: OK

```cl
fn insertion_sort(xs: List[Int]) -> List[Int]
  example insertion_sort([5, 2, 8, 1, 9, 3]) == [1, 2, 3, 5, 8, 9]
  example length(insertion_sort([])) == 0
  var a = xs
  for i in 1 until length(a)
    var key = a[i]
    var j = i
    while j > 0 and a[j - 1] > key
      a[j] = a[j - 1]
      j = j - 1
    end while
    a[j] = key
  end for
  return a
end fn insertion_sort

fn main() uses io
  print(insertion_sort([5, 2, 8, 1, 9, 3]))
end fn main
```

## merge

Task: Write a function that merges two sorted lists into one sorted list. Print the merge of [1, 4, 9] and [2, 3, 10, 11].

### attempt 1: OK

```cl
fn merge(xs: List[Int], ys: List[Int]) -> List[Int]
  example merge(xs: [1, 4, 9], ys: [2, 3, 10, 11]) == [1, 2, 3, 4, 9, 10, 11]
  example merge(xs: [], ys: [5]) == [5]
  example merge(xs: [5], ys: []) == [5]
  example merge(xs: [], ys: []) == []
  var result: List[Int] = []
  var i = 0
  var j = 0
  while i < length(xs) and j < length(ys)
    if xs[i] <= ys[j]
      result = append(list: result, item: xs[i])
      i = i + 1
    else
      result = append(list: result, item: ys[j])
      j = j + 1
    end if
  end while
  while i < length(xs)
    result = append(list: result, item: xs[i])
    i = i + 1
  end while
  while j < length(ys)
    result = append(list: result, item: ys[j])
    j = j + 1
  end while
  return result
end fn merge

fn main() uses io
  print(merge(xs: [1, 4, 9], ys: [2, 3, 10, 11]))
end fn main
```
