## league

Task: Five teams play in a league, listed in this order: Lions, Tigers, Bears, Wolves, Hawks. The results, as home team, home goals, away team, away goals: Lions 2 Tigers 1; Bears 0 Wolves 0; Hawks 3 Lions 3; Lions 1 Bears 3; Tigers 2 Wolves 2; Hawks 1 Bears 1; Wolves 1 Lions 0; Tigers 4 Hawks 1; Bears 1 Tigers 2; Wolves 2 Hawks 4. A win is worth 3 points, a draw 1, a loss 0. Print the standings, one team per line as its name, points and goal difference (goals scored minus goals conceded) separated by single spaces, like `Lions 4 -2`. Order: most points first; on equal points, the larger goal difference first; if that is equal too, more goals scored first.

### attempt 1: OK

```cl
fn better(a: Int, b: Int, pts: List[Int], gd: List[Int], gf: List[Int]) -> Bool
  if pts[a] != pts[b]
    return pts[a] > pts[b]
  end if
  if gd[a] != gd[b]
    return gd[a] > gd[b]
  end if
  return gf[a] > gf[b]
end fn better

fn main() uses io
  let names = ["Lions", "Tigers", "Bears", "Wolves", "Hawks"]
  let home = [0, 2, 4, 0, 1, 4, 3, 1, 2, 3]
  let hg = [2, 0, 3, 1, 2, 1, 1, 4, 1, 2]
  let away = [1, 3, 0, 2, 3, 2, 0, 4, 1, 4]
  let ag = [1, 0, 3, 3, 2, 1, 0, 1, 2, 4]
  var pts = repeat(item: 0, times: 5)
  var gd = repeat(item: 0, times: 5)
  var gf = repeat(item: 0, times: 5)
  for m in 0 until 10
    let h = home[m]
    let a = away[m]
    gf[h] = gf[h] + hg[m]
    gf[a] = gf[a] + ag[m]
    gd[h] = gd[h] + hg[m] - ag[m]
    gd[a] = gd[a] + ag[m] - hg[m]
    if hg[m] > ag[m]
      pts[h] = pts[h] + 3
    else if hg[m] < ag[m]
      pts[a] = pts[a] + 3
    else
      pts[h] = pts[h] + 1
      pts[a] = pts[a] + 1
    end if
  end for
  var order = [0, 1, 2, 3, 4]
  for pass in 0 until 5
    for j in 0 until 4
      if better(a: order[j + 1], b: order[j], pts: pts, gd: gd, gf: gf)
        let tmp = order[j]
        order[j] = order[j + 1]
        order[j + 1] = tmp
      end if
    end for
  end for
  for t in order
    print("{names[t]} {pts[t]} {gd[t]}")
  end for
end fn main
```

## orders

Task: A shop stocks three items, each with a price and a quantity in stock: apple (price 3, 10 in stock), bread (price 5, 2 in stock), cheese (price 12, 1 in stock). Process these orders in turn, each a customer, an item and a quantity: Ann apple 4; Bob bread 3; Cy cheese 1; Ann bread 2; Dee cheese 1; Bob apple 6; Cy fig 1; Ann apple 1. For each order print one line: if the shop doesn't sell the item, `<customer> <item> unknown`; if there isn't enough in stock, `<customer> <item> short` (nothing is sold); otherwise the order is filled, the stock goes down, and the line is `<customer> <item> <quantity> <cost>`, where cost is quantity times price. After all the orders, print each item's remaining stock as `<item> <quantity>` in the order listed above, then each customer's total spending as `<customer> <total>` in the order they first ordered (including customers who bought nothing), then the shop's total revenue as a bare number.

### attempt 1: OK

```cl
struct Item
  name: String
  price: Int
  stock: Int
end struct Item

struct Customer
  name: String
  total: Int
end struct Customer

struct Order
  customer: String
  item: String
  qty: Int
end struct Order

fn find_item(items: List[Item], name: String) -> Int
  for i in 0 until length(items)
    if items[i].name == name
      return i
    end if
  end for
  return -1
end fn find_item

fn find_customer(customers: List[Customer], name: String) -> Int
  for i in 0 until length(customers)
    if customers[i].name == name
      return i
    end if
  end for
  return -1
end fn find_customer

fn main() uses io
  var items: List[Item] = [Item(name: "apple", price: 3, stock: 10), Item(name: "bread", price: 5, stock: 2), Item(name: "cheese", price: 12, stock: 1)]
  var customers: List[Customer] = []
  let orders: List[Order] = [Order(customer: "Ann", item: "apple", qty: 4), Order(customer: "Bob", item: "bread", qty: 3), Order(customer: "Cy", item: "cheese", qty: 1), Order(customer: "Ann", item: "bread", qty: 2), Order(customer: "Dee", item: "cheese", qty: 1), Order(customer: "Bob", item: "apple", qty: 6), Order(customer: "Cy", item: "fig", qty: 1), Order(customer: "Ann", item: "apple", qty: 1)]
  var revenue = 0
  for o in orders
    var ci = find_customer(customers: customers, name: o.customer)
    if ci < 0
      customers = append(list: customers, item: Customer(name: o.customer, total: 0))
      ci = length(customers) - 1
    end if
    let ii = find_item(items: items, name: o.item)
    if ii < 0
      print("{o.customer} {o.item} unknown")
    else if items[ii].stock < o.qty
      print("{o.customer} {o.item} short")
    else
      let cost = o.qty * items[ii].price
      items[ii].stock = items[ii].stock - o.qty
      customers[ci].total = customers[ci].total + cost
      revenue = revenue + cost
      print("{o.customer} {o.item} {o.qty} {cost}")
    end if
  end for
  for it in items
    print("{it.name} {it.stock}")
  end for
  for c in customers
    print("{c.name} {c.total}")
  end for
  print(revenue)
end fn main
```

## life

Task: Run Conway's Game of Life on a grid of 8 rows and 10 columns. Cells beyond the edges count as dead (the grid does not wrap around). Rows and columns are numbered from 0; the live cells at the start, as (row, column), are (0, 1), (1, 2), (2, 0), (2, 1), (2, 2), (5, 9), (6, 9), (7, 9). A live cell with 2 or 3 live neighbours (of its 8) stays alive, a dead cell with exactly 3 becomes alive, and every other cell is dead in the next generation. Run 6 generations. First print one line with the number of live cells after each generation, separated by single spaces. Then print the grid after the last generation, one line per row, `#` for live and `.` for dead.

### attempt 1: OK

```cl
fn neighbors(grid: List[List[Int]], r: Int, c: Int) -> Int
  var count = 0
  for dr in -1 through 1
    for dc in -1 through 1
      if not (dr == 0 and dc == 0)
        let rr = r + dr
        let cc = c + dc
        if rr >= 0 and rr < 8 and cc >= 0 and cc < 10
          count = count + grid[rr][cc]
        end if
      end if
    end for
  end for
  return count
end fn neighbors

fn step(grid: List[List[Int]]) -> List[List[Int]]
  var next = repeat(item: repeat(item: 0, times: 10), times: 8)
  for r in 0 until 8
    for c in 0 until 10
      let n = neighbors(grid: grid, r: r, c: c)
      if grid[r][c] == 1 and (n == 2 or n == 3)
        next[r][c] = 1
      else if grid[r][c] == 0 and n == 3
        next[r][c] = 1
      end if
    end for
  end for
  return next
end fn step

fn live(grid: List[List[Int]]) -> Int
  var total = 0
  for row in grid
    for v in row
      total = total + v
    end for
  end for
  return total
end fn live

fn main() uses io
  var grid = repeat(item: repeat(item: 0, times: 10), times: 8)
  grid[0][1] = 1
  grid[1][2] = 1
  grid[2][0] = 1
  grid[2][1] = 1
  grid[2][2] = 1
  grid[5][9] = 1
  grid[6][9] = 1
  grid[7][9] = 1
  var summary = ""
  for g in 1 through 6
    grid = step(grid)
    let n = live(grid)
    if g == 1
      summary = "{n}"
    else
      summary = "{summary} {n}"
    end if
  end for
  print(summary)
  for r in 0 until 8
    var line = ""
    for c in 0 until 10
      if grid[r][c] == 1
        line = "{line}#"
      else
        line = "{line}."
      end if
    end for
    print(line)
  end for
end fn main
```

## adventure

Task: Write a small text adventure that runs a fixed list of commands. Rooms: Hall, Library, Kitchen, Vault. Exits: from Hall, north goes to Library and east to Kitchen; from Library, south goes to Hall; from Kitchen, west goes to Hall and down to Vault; from Vault, up goes to Kitchen. The way down from Kitchen is locked unless the player carries the key. Items at the start: Library has key, Kitchen has bread, Vault has gold. The player starts in Hall carrying nothing. Commands: `go <direction>` moves and prints the new room's name, or prints `You can't go <direction>.` if there's no such exit, or `The way down is locked.` if it's the locked way and the player has no key. `take <item>` moves the item from the room to the player and prints `Taken: <item>.`, or prints `There is no <item> here.`. `look` prints the room's name, then a line with the room's items separated by single spaces, or `nothing`. `inventory` prints the carried items in the order they were taken, separated by single spaces, or `nothing`. Run these commands in order: go east, take key, take bread, go down, go west, go north, look, take key, go south, go east, go down, take gold, look, go up, go up, inventory.

### attempt 1: OK

```cl
struct State
  room: Int
  rooms: List[List[String]]
  inv: List[String]
end struct State

struct Cmd
  verb: String
  arg: String
end struct Cmd

fn roomName(r: Int) -> String
  example roomName(0) == "Hall"
  example roomName(3) == "Vault"
  if r == 0
    return "Hall"
  else if r == 1
    return "Library"
  else if r == 2
    return "Kitchen"
  else
    return "Vault"
  end if
end fn roomName

fn dest(room: Int, dir: String) -> Int
  example dest(room: 0, dir: "north") == 1
  example dest(room: 1, dir: "east") == -1
  if room == 0
    if dir == "north"
      return 1
    else if dir == "east"
      return 2
    end if
  else if room == 1
    if dir == "south"
      return 0
    end if
  else if room == 2
    if dir == "west"
      return 0
    else if dir == "down"
      return 3
    end if
  else if room == 3
    if dir == "up"
      return 2
    end if
  end if
  return -1
end fn dest

fn has(xs: List[String], item: String) -> Bool
  example has(xs: ["a", "b"], item: "b")
  example not has(xs: ["a"], item: "c")
  for x in xs
    if x == item
      return true
    end if
  end for
  return false
end fn has

fn join(xs: List[String]) -> String
  example join(["a", "b"]) == "a b"
  example join([]) == "nothing"
  if length(xs) == 0
    return "nothing"
  end if
  var line = xs[0]
  for i in 1 until length(xs)
    line = "{line} {xs[i]}"
  end for
  return line
end fn join

fn doGo(s: State, dir: String) -> State uses io
  if s.room == 2 and dir == "down" and not has(xs: s.inv, item: "key")
    print("The way down is locked.")
    return s
  end if
  let d = dest(room: s.room, dir: dir)
  if d < 0
    print("You can't go {dir}.")
    return s
  end if
  var t = s
  t.room = d
  print(roomName(d))
  return t
end fn doGo

fn doTake(s: State, item: String) -> State uses io
  if not has(xs: s.rooms[s.room], item: item)
    print("There is no {item} here.")
    return s
  end if
  var t = s
  var rest: List[String] = []
  for x in s.rooms[s.room]
    if x != item
      rest = append(list: rest, item: x)
    end if
  end for
  t.rooms[t.room] = rest
  t.inv = append(list: t.inv, item: item)
  print("Taken: {item}.")
  return t
end fn doTake

fn main() uses io
  let none: List[String] = []
  let library = ["key"]
  let kitchen = ["bread"]
  let vault = ["gold"]
  let rooms = [none, library, kitchen, vault]
  var s = State(room: 0, rooms: rooms, inv: none)
  let cmds = [Cmd(verb: "go", arg: "east"), Cmd(verb: "take", arg: "key"), Cmd(verb: "take", arg: "bread"), Cmd(verb: "go", arg: "down"), Cmd(verb: "go", arg: "west"), Cmd(verb: "go", arg: "north"), Cmd(verb: "look", arg: ""), Cmd(verb: "take", arg: "key"), Cmd(verb: "go", arg: "south"), Cmd(verb: "go", arg: "east"), Cmd(verb: "go", arg: "down"), Cmd(verb: "take", arg: "gold"), Cmd(verb: "look", arg: ""), Cmd(verb: "go", arg: "up"), Cmd(verb: "go", arg: "up"), Cmd(verb: "inventory", arg: "")]
  for c in cmds
    if c.verb == "go"
      s = doGo(s: s, dir: c.arg)
    else if c.verb == "take"
      s = doTake(s: s, item: c.arg)
    else if c.verb == "look"
      print(roomName(s.room))
      print(join(s.rooms[s.room]))
    else if c.verb == "inventory"
      print(join(s.inv))
    end if
  end for
end fn main
```
