## league

Task: Five teams play in a league, listed in this order: Lions, Tigers, Bears, Wolves, Hawks. The results, as home team, home goals, away team, away goals: Lions 2 Tigers 1; Bears 0 Wolves 0; Hawks 3 Lions 3; Lions 1 Bears 3; Tigers 2 Wolves 2; Hawks 1 Bears 1; Wolves 1 Lions 0; Tigers 4 Hawks 1; Bears 1 Tigers 2; Wolves 2 Hawks 4. A win is worth 3 points, a draw 1, a loss 0. Print the standings, one team per line as its name, points and goal difference (goals scored minus goals conceded) separated by single spaces, like `Lions 4 -2`. Order: most points first; on equal points, the larger goal difference first; if that is equal too, more goals scored first.

### attempt 1: OK

```cl
struct Team
  name: String
  points: Int
  scored: Int
  conceded: Int
end struct Team

fn indexOf(names: List[String], name: String) -> Int
  example indexOf(names: ["Lions", "Tigers"], name: "Tigers") == 1
  for i in 0 until length(names)
    if names[i] == name
      return i
    end if
  end for
  return -1
end fn indexOf

fn ahead(a: Team, b: Team) -> Bool
  if a.points != b.points
    return a.points > b.points
  end if
  if a.scored - a.conceded != b.scored - b.conceded
    return a.scored - a.conceded > b.scored - b.conceded
  end if
  return a.scored > b.scored
end fn ahead

fn main() uses io
  let names = ["Lions", "Tigers", "Bears", "Wolves", "Hawks"]
  let homes = ["Lions", "Bears", "Hawks", "Lions", "Tigers", "Hawks", "Wolves", "Tigers", "Bears", "Wolves"]
  let homeGoals = [2, 0, 3, 1, 2, 1, 1, 4, 1, 2]
  let aways = ["Tigers", "Wolves", "Lions", "Bears", "Wolves", "Bears", "Lions", "Hawks", "Tigers", "Hawks"]
  let awayGoals = [1, 0, 3, 3, 2, 1, 0, 1, 2, 4]

  var teams: List[Team] = []
  for n in names
    teams = append(list: teams, item: Team(name: n, points: 0, scored: 0, conceded: 0))
  end for

  for g in 0 until length(homes)
    let h = indexOf(names: names, name: homes[g])
    let a = indexOf(names: names, name: aways[g])
    let hg = homeGoals[g]
    let ag = awayGoals[g]
    teams[h].scored = teams[h].scored + hg
    teams[h].conceded = teams[h].conceded + ag
    teams[a].scored = teams[a].scored + ag
    teams[a].conceded = teams[a].conceded + hg
    if hg > ag
      teams[h].points = teams[h].points + 3
    else if hg < ag
      teams[a].points = teams[a].points + 3
    else
      teams[h].points = teams[h].points + 1
      teams[a].points = teams[a].points + 1
    end if
  end for

  for p in 0 until 4
    for j in 0 until 4
      if ahead(a: teams[j + 1], b: teams[j])
        let tmp = teams[j]
        teams[j] = teams[j + 1]
        teams[j + 1] = tmp
      end if
    end for
  end for

  for t in teams
    print("{t.name} {t.points} {t.scored - t.conceded}")
  end for
end fn main
```

## orders

Task: A shop stocks three items, each with a price and a quantity in stock: apple (price 3, 10 in stock), bread (price 5, 2 in stock), cheese (price 12, 1 in stock). Process these orders in turn, each a customer, an item and a quantity: Ann apple 4; Bob bread 3; Cy cheese 1; Ann bread 2; Dee cheese 1; Bob apple 6; Cy fig 1; Ann apple 1. For each order print one line: if the shop doesn't sell the item, `<customer> <item> unknown`; if there isn't enough in stock, `<customer> <item> short` (nothing is sold); otherwise the order is filled, the stock goes down, and the line is `<customer> <item> <quantity> <cost>`, where cost is quantity times price. After all the orders, print each item's remaining stock as `<item> <quantity>` in the order listed above, then each customer's total spending as `<customer> <total>` in the order they first ordered (including customers who bought nothing), then the shop's total revenue as a bare number.

### attempt 1: OK

```cl
struct Item
  name: String
  price: Int
  stock: Int
end struct Item

struct Order
  customer: String
  product: String
  qty: Int
end struct Order

struct Spend
  customer: String
  total: Int
end struct Spend

fn find_item(things: List[Item], wanted: String) -> Option[Int]
  for i in 0 until length(things)
    if things[i].name == wanted
      return Some(i)
    end if
  end for
  return None
end fn find_item

fn find_spend(spends: List[Spend], wanted: String) -> Option[Int]
  for i in 0 until length(spends)
    if spends[i].customer == wanted
      return Some(i)
    end if
  end for
  return None
end fn find_spend

fn add_spend(spends: List[Spend], wanted: String, amount: Int) -> List[Spend]
  var updated = spends
  match find_spend(spends: spends, wanted: wanted)
    case Some(i)
      updated[i].total = updated[i].total + amount
    case None
      updated = append(list: updated, item: Spend(customer: wanted, total: amount))
  end match
  return updated
end fn add_spend

fn main() uses io
  var items: List[Item] = [Item(name: "apple", price: 3, stock: 10), Item(name: "bread", price: 5, stock: 2), Item(name: "cheese", price: 12, stock: 1)]
  let orders: List[Order] = [Order(customer: "Ann", product: "apple", qty: 4), Order(customer: "Bob", product: "bread", qty: 3), Order(customer: "Cy", product: "cheese", qty: 1), Order(customer: "Ann", product: "bread", qty: 2), Order(customer: "Dee", product: "cheese", qty: 1), Order(customer: "Bob", product: "apple", qty: 6), Order(customer: "Cy", product: "fig", qty: 1), Order(customer: "Ann", product: "apple", qty: 1)]
  var spends: List[Spend] = []
  var revenue = 0
  for o in orders
    spends = add_spend(spends: spends, wanted: o.customer, amount: 0)
    match find_item(things: items, wanted: o.product)
      case None
        print("{o.customer} {o.product} unknown")
      case Some(ix)
        if items[ix].stock < o.qty
          print("{o.customer} {o.product} short")
        else
          let cost = o.qty * items[ix].price
          items[ix].stock = items[ix].stock - o.qty
          revenue = revenue + cost
          spends = add_spend(spends: spends, wanted: o.customer, amount: cost)
          print("{o.customer} {o.product} {o.qty} {cost}")
        end if
    end match
  end for
  for thing in items
    print("{thing.name} {thing.stock}")
  end for
  for sp in spends
    print("{sp.customer} {sp.total}")
  end for
  print(revenue)
end fn main
```

## life

Task: Run Conway's Game of Life on a grid of 8 rows and 10 columns. Cells beyond the edges count as dead (the grid does not wrap around). Rows and columns are numbered from 0; the live cells at the start, as (row, column), are (0, 1), (1, 2), (2, 0), (2, 1), (2, 2), (5, 9), (6, 9), (7, 9). A live cell with 2 or 3 live neighbours (of its 8) stays alive, a dead cell with exactly 3 becomes alive, and every other cell is dead in the next generation. Run 6 generations. First print one line with the number of live cells after each generation, separated by single spaces. Then print the grid after the last generation, one line per row, `#` for live and `.` for dead.

### attempt 1: OK

```cl
fn make_grid(rows: Int, cols: Int) -> List[List[Int]]
  example make_grid(rows: 2, cols: 3) == [[0, 0, 0], [0, 0, 0]]
  var g: List[List[Int]] = []
  for r in 0 until rows
    g = append(list: g, item: repeat(item: 0, times: cols))
  end for
  return g
end fn make_grid

fn live_at(g: List[List[Int]], r: Int, c: Int) -> Int
  if r < 0 or r >= length(g) or c < 0 or c >= length(g[0])
    return 0
  end if
  return g[r][c]
end fn live_at

fn neighbours(g: List[List[Int]], r: Int, c: Int) -> Int
  var n = 0
  for dr in -1 through 1
    for dc in -1 through 1
      if not (dr == 0 and dc == 0)
        n = n + live_at(g: g, r: r + dr, c: c + dc)
      end if
    end for
  end for
  return n
end fn neighbours

fn step(g: List[List[Int]]) -> List[List[Int]]
  var nxt: List[List[Int]] = []
  for r in 0 until length(g)
    var row: List[Int] = []
    for c in 0 until length(g[r])
      var k = neighbours(g: g, r: r, c: c)
      var v = 0
      if g[r][c] == 1
        if k == 2 or k == 3
          v = 1
        end if
      else if k == 3
        v = 1
      end if
      row = append(list: row, item: v)
    end for
    nxt = append(list: nxt, item: row)
  end for
  return nxt
end fn step

fn count_live(g: List[List[Int]]) -> Int
  example count_live([[1, 0], [1, 1]]) == 3
  var total = 0
  for row in g
    for x in row
      total = total + x
    end for
  end for
  return total
end fn count_live

fn render_row(row: List[Int]) -> String
  example render_row([1, 0, 1]) == "#.#"
  var line = ""
  for x in row
    if x == 1
      line = "{line}#"
    else
      line = "{line}."
    end if
  end for
  return line
end fn render_row

fn main() uses io
  var g = make_grid(rows: 8, cols: 10)
  g[0][1] = 1
  g[1][2] = 1
  g[2][0] = 1
  g[2][1] = 1
  g[2][2] = 1
  g[5][9] = 1
  g[6][9] = 1
  g[7][9] = 1

  var line = ""
  for gen in 1 through 6
    g = step(g: g)
    if gen == 1
      line = "{count_live(g)}"
    else
      line = "{line} {count_live(g)}"
    end if
  end for
  print(line)

  for row in g
    print(render_row(row))
  end for
end fn main
```

## adventure

Task: Write a small text adventure that runs a fixed list of commands. Rooms: Hall, Library, Kitchen, Vault. Exits: from Hall, north goes to Library and east to Kitchen; from Library, south goes to Hall; from Kitchen, west goes to Hall and down to Vault; from Vault, up goes to Kitchen. The way down from Kitchen is locked unless the player carries the key. Items at the start: Library has key, Kitchen has bread, Vault has gold. The player starts in Hall carrying nothing. Commands: `go <direction>` moves and prints the new room's name, or prints `You can't go <direction>.` if there's no such exit, or `The way down is locked.` if it's the locked way and the player has no key. `take <item>` moves the item from the room to the player and prints `Taken: <item>.`, or prints `There is no <item> here.`. `look` prints the room's name, then a line with the room's items separated by single spaces, or `nothing`. `inventory` prints the carried items in the order they were taken, separated by single spaces, or `nothing`. Run these commands in order: go east, take key, take bread, go down, go west, go north, look, take key, go south, go east, go down, take gold, look, go up, go up, inventory.

### attempt 1: OK

```cl
enum Command
  Go(dir: String)
  Take(item: String)
  Look
  Inventory
end enum Command

fn exit_to(room: Int, dir: String) -> Option[Int]
  example exit_to(room: 0, dir: "north") == Some(1)
  example exit_to(room: 3, dir: "up") == Some(2)
  example exit_to(room: 1, dir: "east") == None
  if room == 0
    if dir == "north"
      return Some(1)
    end if
    if dir == "east"
      return Some(2)
    end if
  end if
  if room == 1
    if dir == "south"
      return Some(0)
    end if
  end if
  if room == 2
    if dir == "west"
      return Some(0)
    end if
    if dir == "down"
      return Some(3)
    end if
  end if
  if room == 3
    if dir == "up"
      return Some(2)
    end if
  end if
  return None
end fn exit_to

fn contains(xs: List[String], x: String) -> Bool
  example contains(xs: ["a", "b"], x: "b") == true
  example contains(xs: [], x: "a") == false
  for s in xs
    if s == x
      return true
    end if
  end for
  return false
end fn contains

fn without(xs: List[String], x: String) -> List[String]
  example without(xs: ["a", "b", "c"], x: "b") == ["a", "c"]
  var result: List[String] = []
  for s in xs
    if s != x
      result = append(list: result, item: s)
    end if
  end for
  return result
end fn without

fn join_words(xs: List[String]) -> String
  example join_words(xs: ["a", "b"]) == "a b"
  example join_words(xs: ["a"]) == "a"
  example join_words(xs: []) == ""
  var text = ""
  for i in 0 until length(xs)
    if i == 0
      text = xs[i]
    else
      text = "{text} {xs[i]}"
    end if
  end for
  return text
end fn join_words

fn main() uses io
  let names: List[String] = ["Hall", "Library", "Kitchen", "Vault"]
  var items: List[List[String]] = [[], ["key"], ["bread"], ["gold"]]
  var inv: List[String] = []
  var room = 0
  let cmds: List[Command] = [Go("east"), Take("key"), Take("bread"), Go("down"), Go("west"), Go("north"), Look, Take("key"), Go("south"), Go("east"), Go("down"), Take("gold"), Look, Go("up"), Go("up"), Inventory]
  for c in cmds
    match c
      case Go(d)
        let dest = exit_to(room: room, dir: d)
        match dest
          case Some(target)
            if target == 3 and room == 2 and not contains(xs: inv, x: "key")
              print("The way down is locked.")
            else
              room = target
              print(names[room])
            end if
          case None
            print("You can't go {d}.")
        end match
      case Take(it)
        if contains(xs: items[room], x: it)
          items[room] = without(xs: items[room], x: it)
          inv = append(list: inv, item: it)
          print("Taken: {it}.")
        else
          print("There is no {it} here.")
        end if
      case Look
        print(names[room])
        if length(items[room]) == 0
          print("nothing")
        else
          print(join_words(xs: items[room]))
        end if
      case Inventory
        if length(inv) == 0
          print("nothing")
        else
          print(join_words(xs: inv))
        end if
    end match
  end for
end fn main
```
