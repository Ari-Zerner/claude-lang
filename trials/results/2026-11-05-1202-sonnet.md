## fib

Task: Print the first 15 Fibonacci numbers, starting 0, 1, one per line.

### attempt 1: OK

```cl
fn fib(n: Int) -> Int
  example fib(0) == 0
  example fib(1) == 1
  example fib(10) == 55
  var a = 0
  var b = 1
  var i = 0
  while i < n
    let t = a + b
    a = b
    b = t
    i = i + 1
  end while
  return a
end fn fib

fn main() uses io
  var k = 0
  while k < 15
    print(fib(k))
    k = k + 1
  end while
end fn main
```

## primes

fn is_prime(n: Int) -> Bool
  example is_prime(2) == true
  example is_prime(1) == false
  example is_prime(9) == false
  example is_prime(13) == true
  if n < 2
    return false
  end if
  var d = 2
  while d * d <= n
    if n % d == 0
      return false
    end if
    d = d + 1
  end while
  return true
end fn is_prime

fn main() uses io
  var i = 2
  while i < 50
    if is_prime(i)
      print(i)
    end if
    i = i + 1
  end while
end fn main

## collatz

Task: Print how many steps the Collatz process (n -> n/2 if even, 3n+1 if odd) takes to reach 1 starting from 27.

### attempt 1: OK

```cl
fn collatz_steps(start: Int) -> Int
  example collatz_steps(1) == 0
  example collatz_steps(2) == 1
  example collatz_steps(6) == 8
  example collatz_steps(27) == 111
  var n = start
  var steps = 0
  while n != 1
    if n % 2 == 0
      n = n / 2
    else
      n = 3 * n + 1
    end if
    steps = steps + 1
  end while
  return steps
end fn collatz_steps

fn main() uses io
  print(collatz_steps(27))
end fn main
```

## isqrt

Task: Print the integer square root (the floor of the square root) of 1000000007, computed with binary search.

### attempt 1: OK

```cl
# Integer square root by binary search.
# Invariant: lo*lo <= n and hi*hi > n (for n < 100000*100000).
fn isqrt(n: Int) -> Int
  example isqrt(0) == 0
  example isqrt(1) == 1
  example isqrt(16) == 4
  example isqrt(99) == 9
  example isqrt(1000000007) == 31622
  var lo = 0
  var hi = 100000
  while hi - lo > 1
    let mid = (lo + hi) / 2
    if mid * mid <= n
      lo = mid
    else
      hi = mid
    end if
  end while
  return lo
end fn isqrt

fn main() uses io
  print(isqrt(1000000007))
end fn main
```

## digits

Task: Print the sum of the decimal digits of 2 to the power 20.

### attempt 1: OK

```cl
fn pow2(n: Int) -> Int
  example pow2(0) == 1
  example pow2(10) == 1024
  var result = 1
  var i = 0
  while i < n
    result = result * 2
    i = i + 1
  end while
  return result
end fn pow2

fn digit_sum(n: Int) -> Int
  example digit_sum(0) == 0
  example digit_sum(1048576) == 31
  var total = 0
  var rest = n
  while rest > 0
    total = total + rest % 10
    rest = rest / 10
  end while
  return total
end fn digit_sum

fn main() uses io
  print(digit_sum(pow2(20)))
end fn main
```

## perfect

Task: A perfect number equals the sum of its proper divisors. Print whether 28 is perfect, then whether 12 is perfect (one result per line).

### attempt 1: OK

```cl
fn divisor_sum(n: Int) -> Int
  example divisor_sum(6) == 6
  example divisor_sum(28) == 28
  example divisor_sum(12) == 16
  var total = 0
  var i = 1
  while i < n
    if n % i == 0
      total = total + i
    end if
    i = i + 1
  end while
  return total
end fn divisor_sum

fn is_perfect(n: Int) -> Bool
  example is_perfect(28) == true
  example is_perfect(12) == false
  example is_perfect(6) == true
  return divisor_sum(n) == n
end fn is_perfect

fn main() uses io
  print(is_perfect(28))
  print(is_perfect(12))
end fn main
```
