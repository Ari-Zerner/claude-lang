## fib-list

Task: Build a list of the first 10 Fibonacci numbers (starting 0, 1) and print the list.

### attempt 1: OK

```cl
fn fibs(n: Int) -> List[Int]
  example fibs(0) == []
  example fibs(1) == [0]
  example fibs(5) == [0, 1, 1, 2, 3]
  var result: List[Int] = []
  var a = 0
  var b = 1
  var i = 0
  while i < n
    result = append(list: result, item: a)
    let next = a + b
    a = b
    b = next
    i = i + 1
  end while
  return result
end fn fibs

fn main() uses io
  print(fibs(10))
end fn main
```

## words

Task: Given the list of words apple, kiwi, banana, fig: print the longest word, then the total number of characters in all the words.

### attempt 1: OK

```cl
fn longest(words: List[String]) -> String
  example longest(["apple", "kiwi", "banana", "fig"]) == "banana"
  example longest(["a", "bb"]) == "bb"
  var best = words[0]
  var i = 1
  while i < length(words)
    if length(words[i]) > length(best)
      best = words[i]
    end if
    i = i + 1
  end while
  return best
end fn longest

fn totalChars(words: List[String]) -> Int
  example totalChars(["apple", "kiwi", "banana", "fig"]) == 18
  example totalChars([]) == 0
  var sum = 0
  var i = 0
  while i < length(words)
    sum = sum + length(words[i])
    i = i + 1
  end while
  return sum
end fn totalChars

fn main() uses io
  let words = ["apple", "kiwi", "banana", "fig"]
  print(longest(words))
  print(totalChars(words))
end fn main
```

fn isort(xs: List[Int]) -> List[Int]
  example isort([3, 1, 2]) == [1, 2, 3]
  example isort([5, 5, 1]) == [1, 5, 5]
  var ys = xs
  var i = 1

## merge

Task: Write a function that merges two sorted lists into one sorted list. Print the merge of [1, 4, 9] and [2, 3, 10, 11].

### attempt 1: OK

```cl
fn merge(a: List[Int], b: List[Int]) -> List[Int]
  example merge(a: [1, 4, 9], b: [2, 3, 10, 11]) == [1, 2, 3, 4, 9, 10, 11]
  example merge(a: [5], b: [1, 2]) == [1, 2, 5]
  var out: List[Int] = []
  var i = 0
  var j = 0
  while i < length(a) and j < length(b)
    if a[i] <= b[j]
      out = append(list: out, item: a[i])
      i = i + 1
    else
      out = append(list: out, item: b[j])
      j = j + 1
    end if
  end while
  while i < length(a)
    out = append(list: out, item: a[i])
    i = i + 1
  end while
  while j < length(b)
    out = append(list: out, item: b[j])
    j = j + 1
  end while
  return out
end fn merge

fn main() uses io
  let left = [1, 4, 9]
  let right = [2, 3, 10, 11]
  let merged = merge(a: left, b: right)
  print(merged)
end fn main
```
