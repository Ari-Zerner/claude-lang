## first-big-fib

Task: Print the first Fibonacci number that has more than 10 decimal digits.

### attempt 1: OK

```cl
fn first_big_fib() -> Int
  var a = 0
  var b = 1
  while b < 10_000_000_000
    let next = a + b
    a = b
    b = next
  end while
  return b
end fn first_big_fib

fn main() uses io
  print(first_big_fib())
end fn main
```

## triangle

Task: The n-th triangular number is 1 + 2 + ... + n. Print the smallest n whose triangular number has more than 50 divisors.

### attempt 1: OK

```cl
fn triangle(n: Int) -> Int
  example triangle(1) == 1
  example triangle(4) == 10
  return (n * (n + 1)) / 2
end fn triangle

fn count_divisors(n: Int) -> Int
  example count_divisors(1) == 1
  example count_divisors(28) == 6
  example count_divisors(36) == 9
  var count = 0
  var i = 1
  while i * i <= n
    if n % i == 0
      if i * i == n
        count = count + 1
      else
        count = count + 2
      end if
    end if
    i = i + 1
  end while
  return count
end fn count_divisors

fn main() uses io
  var n = 1
  while count_divisors(triangle(n)) <= 50
    n = n + 1
  end while
  print(n)
end fn main
```

## next-prime

Task: Print the first prime number greater than 10000.

### attempt 1: OK

```cl
fn is_prime(n: Int) -> Bool
  example is_prime(2) == true
  example is_prime(1) == false
  example is_prime(9) == false
  example is_prime(10007) == true
  if n < 2
    return false
  end if
  var i = 2
  while i * i <= n
    if n % i == 0
      return false
    end if
    i = i + 1
  end while
  return true
end fn is_prime

fn main() uses io
  var n = 10001
  while not is_prime(n)
    n = n + 1
  end while
  print(n)
end fn main
```

## even-odd

Task: Write two mutually recursive functions is_even and is_odd that don't use `%`, `/` or `*`. Print is_even(10), then is_even(7).

### attempt 1: OK

```cl
fn is_even(n: Int) -> Bool
  example is_even(0) == true
  example is_even(10) == true
  example is_even(7) == false
  if n < 0
    return is_even(-n)
  end if
  if n == 0
    return true
  end if
  return is_odd(n - 1)
end fn is_even

fn is_odd(n: Int) -> Bool
  example is_odd(0) == false
  example is_odd(7) == true
  example is_odd(10) == false
  if n < 0
    return is_odd(-n)
  end if
  if n == 0
    return false
  end if
  return is_even(n - 1)
end fn is_odd

fn main() uses io
  print(is_even(10))
  print(is_even(7))
end fn main
```

## clamp

### attempt 1: OK

```cl
fn clamp(value: Int, low: Int, high: Int) -> Int
  example clamp(value: 15, low: 0, high: 10) == 10
  example clamp(value: -5, low: 0, high: 10) == 0
  example clamp(value: 5, low: 0, high: 10) == 5

    return low
```

## countdown

Task: Write a function that prints the numbers from n down to 1, one per line. From main, call it with 5 and then print 0.

### attempt 1: OK

```cl
fn countdown(n: Int) uses io
  var i = n
  while i >= 1
    print(i)
    i = i - 1
  end while
```

## coins

Task: Print the number of different ways to make 100 cents from coins of 1, 5, 10, 25 and 50 cents (order doesn't matter).

### attempt 1: OK

```cl
fn coin(k: Int) -> Int
  example coin(1) == 1
  example coin(3) == 10
  example coin(5) == 50
  if k == 1
    return 1
  else if k == 2
    return 5
  else if k == 3
    return 10
  else if k == 4
    return 25
  else
    return 50
  end if
end fn coin

fn ways(amount: Int, kind: Int) -> Int
  example ways(amount: 0, kind: 5) == 1
  example ways(amount: 10, kind: 3) == 4
  example ways(amount: 5, kind: 0) == 0
  if amount == 0
    return 1
  end if
  if amount < 0 or kind == 0
    return 0
  end if
  return ways(amount: amount - coin(kind), kind: kind) + ways(amount: amount, kind: kind - 1)
end fn ways

fn main() uses io
  print(ways(amount: 100, kind: 5))
end fn main
```

## table

Task: Print a multiplication table for 1 to 4: four lines, each with the four products separated by spaces.

### attempt 1: compiled (open-ended task)

```cl
# Note: print only outputs one Int or Bool plus a newline, and the language has
# no strings, so the products can't be put on one line separated by spaces.
# This prints the 16 products in row order, one per line.

  var i = 1
  while i <= 4
    var j = 1
    while j <= 4
      print(i * j)
      j = j + 1

    i = i + 1
```
