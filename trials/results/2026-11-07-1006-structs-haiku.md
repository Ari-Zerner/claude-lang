## scores

Task: Keep a list of players, each with a name and a score: Ann, Bob and Cy, all starting at 0. Add 5 to Bob's score twice and 3 to Cy's score once, then print each player as their name and score, like `Ann 0`, one per line.

### attempt 1: OK

```cl
struct Player
  name: String
  score: Int
end struct Player

fn main() uses io
  var ps: List[Player] = [Player(name: "Ann", score: 0), Player(name: "Bob", score: 0), Player(name: "Cy", score: 0)]
  ps[1].score = ps[1].score + 5
  ps[1].score = ps[1].score + 5
  ps[2].score = ps[2].score + 3
  for p in ps
    print("{p.name} {p.score}")
  end for
end fn main
```

## segment

Task: A line segment has a start point and an end point. Write a function that returns a segment's squared length. Make the segment from (3, 4) to (6, 8) and print its squared length; then change its start point's x to 0 and print its squared length again.

### attempt 1: OK

```cl
struct Point
  x: Int
  y: Int
end struct Point

struct Segment
  start: Point
  finish: Point
end struct Segment

fn squared_length(s: Segment) -> Int
  example squared_length(Segment(start: Point(x: 3, y: 4), finish: Point(x: 6, y: 8))) == 25
  example squared_length(Segment(start: Point(x: 0, y: 0), finish: Point(x: 0, y: 0))) == 0
  let dx = s.finish.x - s.start.x
  let dy = s.finish.y - s.start.y
  return dx * dx + dy * dy
end fn squared_length

fn main() uses io
  var s = Segment(start: Point(x: 3, y: 4), finish: Point(x: 6, y: 8))
  print(squared_length(s))
  s.start.x = 0
  print(squared_length(s))
end fn main
```
