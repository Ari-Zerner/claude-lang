## triangle

Task: The n-th triangular number is 1 + 2 + ... + n. Print the smallest n whose triangular number has more than 50 divisors.

### attempt 1: OK

```cl
fn triangular(n: Int) -> Int
  example triangular(n: 1) == 1
  example triangular(n: 4) == 10
  return (n * (n + 1)) / 2
end fn triangular

fn divisors(t: Int) -> Int
  example divisors(t: 1) == 1
  example divisors(t: 12) == 6
  example divisors(t: 36) == 9
  var count = 0
  var d = 1
  while d * d <= t
    if t % d == 0
      if d * d == t
        count = count + 1
      else
        count = count + 2
      end if
    end if
    d = d + 1
  end while
  return count
end fn divisors

fn main() uses io
  var n = 1
  var found = false
  while not found
    if divisors(t: triangular(n: n)) > 50
      found = true
    else
      n = n + 1
    end if
  end while
  print(n)
end fn main
```

## next-prime

Task: Print the first prime number greater than 10000.

### attempt 1: OK

```cl
fn is_prime(n: Int) -> Bool
  example is_prime(n: 1) == false
  example is_prime(n: 2) == true
  example is_prime(n: 9) == false
  example is_prime(n: 10007) == true
  if n < 2
    return false
  end if
  var d = 2
  while d * d <= n
    if n % d == 0
      return false
    end if
    d = d + 1
  end while
  return true
end fn is_prime

fn main() uses io
  var n = 10001
  while is_prime(n) == false
    n = n + 1
  end while
  print(n)
end fn main
```

## even-odd

Task: Write two mutually recursive functions is_even and is_odd that don't use `%`, `/` or `*`. Print is_even(10), then is_even(7).

### attempt 1: OK

```cl
fn is_even(n: Int) -> Bool
  example is_even(0) == true
  example is_even(7) == false
  if n == 0
    return true
  else
    return is_odd(n: n - 1)
  end if
end fn is_even

fn is_odd(n: Int) -> Bool
  example is_odd(1) == true
  example is_odd(4) == false
  if n == 0
    return false
  else
    return is_even(n: n - 1)
  end if
end fn is_odd

fn main() uses io
  print(is_even(10))
  print(is_even(7))
end fn main
```

## clamp

Task: Write a function clamp that takes a value, a lower bound and an upper bound. Print the result of clamping 15, then -5, then 5, each to the range 0 to 10.

### attempt 1: OK

```cl
fn clamp(value: Int, lo: Int, hi: Int) -> Int
  example clamp(value: 15, lo: 0, hi: 10) == 10
  example clamp(value: -5, lo: 0, hi: 10) == 0
  example clamp(value: 5, lo: 0, hi: 10) == 5
  if value < lo
    return lo
  else if value > hi
    return hi
  else
    return value
  end if
```

## countdown

Task: Write a function that prints the numbers from n down to 1, one per line. From main, call it with 5 and then print 0.

### attempt 1: OK

```cl
fn countdown(n: Int) uses io
  var i = n
  while i >= 1
    print(i)
    i = i - 1
  end while
end fn countdown
```

## coins

Task: Print the number of different ways to make 100 cents from coins of 1, 5, 10, 25 and 50 cents (order doesn't matter).

### attempt 1: OK

```cl
fn count_ways() -> Int
  example count_ways() == 292
  var ways = 0
  var a = 0
  while a <= 2
    var b = 0
    while b <= 4
      var c = 0
      while c <= 10
        var d = 0
        while d <= 20
          if 50 * a + 25 * b + 10 * c + 5 * d <= 100
            ways = ways + 1
          end if
          d = d + 1
        end while
        c = c + 1
      end while
      b = b + 1
    end while
    a = a + 1
  end while
  return ways
end fn count_ways

fn main() uses io
  print(count_ways())
end fn main
```
