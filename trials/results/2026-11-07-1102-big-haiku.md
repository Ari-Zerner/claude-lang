## league

Task: Five teams play in a league, listed in this order: Lions, Tigers, Bears, Wolves, Hawks. The results, as home team, home goals, away team, away goals: Lions 2 Tigers 1; Bears 0 Wolves 0; Hawks 3 Lions 3; Lions 1 Bears 3; Tigers 2 Wolves 2; Hawks 1 Bears 1; Wolves 1 Lions 0; Tigers 4 Hawks 1; Bears 1 Tigers 2; Wolves 2 Hawks 4. A win is worth 3 points, a draw 1, a loss 0. Print the standings, one team per line as its name, points and goal difference (goals scored minus goals conceded) separated by single spaces, like `Lions 4 -2`. Order: most points first; on equal points, the larger goal difference first; if that is equal too, more goals scored first.

### attempt 1: OK

```cl
struct Team
  name: String
  points: Int
  scored: Int
  conceded: Int
end struct Team

fn new_team(name: String) -> Team
  return Team(name: name, points: 0, scored: 0, conceded: 0)
end fn new_team

fn index_of(teams: List[Team], name: String) -> Int
  example index_of(teams: [new_team(name: "A"), new_team(name: "B")], name: "B") == 1
  var found = -1
  for i in 0 until length(teams)
    if teams[i].name == name
      found = i
    end if
  end for
  return found
end fn index_of

fn goal_diff(t: Team) -> Int
  example goal_diff(Team(name: "A", points: 0, scored: 5, conceded: 7)) == -2
  return t.scored - t.conceded
end fn goal_diff

fn better(a: Team, b: Team) -> Bool
  if a.points != b.points
    return a.points > b.points
  end if
  if goal_diff(a) != goal_diff(b)
    return goal_diff(a) > goal_diff(b)
  end if
  return a.scored > b.scored
end fn better

fn apply(teams: List[Team], home: String, hg: Int, away: String, ag: Int) -> List[Team]
  var ts = teams
  var h = index_of(teams: ts, name: home)
  var a = index_of(teams: ts, name: away)
  ts[h].scored = ts[h].scored + hg
  ts[h].conceded = ts[h].conceded + ag
  ts[a].scored = ts[a].scored + ag
  ts[a].conceded = ts[a].conceded + hg
  if hg > ag
    ts[h].points = ts[h].points + 3
  else if hg == ag
    ts[h].points = ts[h].points + 1
    ts[a].points = ts[a].points + 1
  else
    ts[a].points = ts[a].points + 3
  end if
  return ts
end fn apply

fn main() uses io
  var teams: List[Team] = [new_team(name: "Lions"), new_team(name: "Tigers"), new_team(name: "Bears"), new_team(name: "Wolves"), new_team(name: "Hawks")]

  teams = apply(teams: teams, home: "Lions", hg: 2, away: "Tigers", ag: 1)
  teams = apply(teams: teams, home: "Bears", hg: 0, away: "Wolves", ag: 0)
  teams = apply(teams: teams, home: "Hawks", hg: 3, away: "Lions", ag: 3)
  teams = apply(teams: teams, home: "Lions", hg: 1, away: "Bears", ag: 3)
  teams = apply(teams: teams, home: "Tigers", hg: 2, away: "Wolves", ag: 2)
  teams = apply(teams: teams, home: "Hawks", hg: 1, away: "Bears", ag: 1)
  teams = apply(teams: teams, home: "Wolves", hg: 1, away: "Lions", ag: 0)
  teams = apply(teams: teams, home: "Tigers", hg: 4, away: "Hawks", ag: 1)
  teams = apply(teams: teams, home: "Bears", hg: 1, away: "Tigers", ag: 2)
  teams = apply(teams: teams, home: "Wolves", hg: 2, away: "Hawks", ag: 4)

  var order = teams
  for i in 1 until length(order)
    var j = i
    while j > 0 and better(a: order[j], b: order[j - 1])
      var tmp = order[j]
      order[j] = order[j - 1]
      order[j - 1] = tmp
      j = j - 1
    end while
  end for

  for t in order
    print("{t.name} {t.points} {goal_diff(t)}")
  end for
end fn main
```

## orders

Task: A shop stocks three items, each with a price and a quantity in stock: apple (price 3, 10 in stock), bread (price 5, 2 in stock), cheese (price 12, 1 in stock). Process these orders in turn, each a customer, an item and a quantity: Ann apple 4; Bob bread 3; Cy cheese 1; Ann bread 2; Dee cheese 1; Bob apple 6; Cy fig 1; Ann apple 1. For each order print one line: if the shop doesn't sell the item, `<customer> <item> unknown`; if there isn't enough in stock, `<customer> <item> short` (nothing is sold); otherwise the order is filled, the stock goes down, and the line is `<customer> <item> <quantity> <cost>`, where cost is quantity times price. After all the orders, print each item's remaining stock as `<item> <quantity>` in the order listed above, then each customer's total spending as `<customer> <total>` in the order they first ordered (including customers who bought nothing), then the shop's total revenue as a bare number.

### attempt 1: OK

```cl
struct Order
  customer: String
  item: String
  qty: Int
end struct Order

fn find(names: List[String], name: String) -> Int
  example find(names: ["apple", "bread", "cheese"], name: "bread") == 1
  example find(names: ["apple", "bread", "cheese"], name: "fig") == -1
  for i in 0 until length(names)
    if names[i] == name
      return i
    end if
  end for
  return -1
end fn find

fn main() uses io
  let itemNames = ["apple", "bread", "cheese"]
  let prices = [3, 5, 12]
  var stock = [10, 2, 1]
  let orders = [Order(customer: "Ann", item: "apple", qty: 4), Order(customer: "Bob", item: "bread", qty: 3), Order(customer: "Cy", item: "cheese", qty: 1), Order(customer: "Ann", item: "bread", qty: 2), Order(customer: "Dee", item: "cheese", qty: 1), Order(customer: "Bob", item: "apple", qty: 6), Order(customer: "Cy", item: "fig", qty: 1), Order(customer: "Ann", item: "apple", qty: 1)]
  var custNames: List[String] = []
  var custTotals: List[Int] = []
  var revenue = 0
  for o in orders
    var ci = find(names: custNames, name: o.customer)
    if ci == -1
      custNames = append(list: custNames, item: o.customer)
      custTotals = append(list: custTotals, item: 0)
      ci = length(custNames) - 1
    end if
    let ii = find(names: itemNames, name: o.item)
    if ii == -1
      print("{o.customer} {o.item} unknown")
    else if o.qty > stock[ii]
      print("{o.customer} {o.item} short")
    else
      let cost = prices[ii] * o.qty
      stock[ii] = stock[ii] - o.qty
      custTotals[ci] = custTotals[ci] + cost
      revenue = revenue + cost
      print("{o.customer} {o.item} {o.qty} {cost}")
    end if
  end for
  for s in 0 until length(itemNames)
    print("{itemNames[s]} {stock[s]}")
  end for
  for c in 0 until length(custNames)
    print("{custNames[c]} {custTotals[c]}")
  end for
  print("{revenue}")
end fn main
```

## life

Task: Run Conway's Game of Life on a grid of 8 rows and 10 columns. Cells beyond the edges count as dead (the grid does not wrap around). Rows and columns are numbered from 0; the live cells at the start, as (row, column), are (0, 1), (1, 2), (2, 0), (2, 1), (2, 2), (5, 9), (6, 9), (7, 9). A live cell with 2 or 3 live neighbours (of its 8) stays alive, a dead cell with exactly 3 becomes alive, and every other cell is dead in the next generation. Run 6 generations. First print one line with the number of live cells after each generation, separated by single spaces. Then print the grid after the last generation, one line per row, `#` for live and `.` for dead.

### attempt 1: OK

```cl
fn empty_grid() -> List[List[Int]]
  var g: List[List[Int]] = []
  for r in 0 until 8
    g = append(list: g, item: repeat(item: 0, times: 10))
  end for
  return g
end fn empty_grid

fn cell_at(g: List[List[Int]], r: Int, c: Int) -> Int
  if r < 0 or r > 7 or c < 0 or c > 9
    return 0
  end if
  return g[r][c]
end fn cell_at

fn neighbours(g: List[List[Int]], r: Int, c: Int) -> Int
  var n = 0
  for dr in -1 through 1
    for dc in -1 through 1
      if dr != 0 or dc != 0
        n = n + cell_at(g: g, r: r + dr, c: c + dc)
      end if
    end for
  end for
  return n
end fn neighbours

fn next_cell(alive: Int, n: Int) -> Int
  example next_cell(alive: 1, n: 2) == 1
  example next_cell(alive: 1, n: 3) == 1
  example next_cell(alive: 1, n: 1) == 0
  example next_cell(alive: 0, n: 3) == 1
  example next_cell(alive: 0, n: 2) == 0
  if alive == 1
    if n == 2 or n == 3
      return 1
    end if
    return 0
  end if
  if n == 3
    return 1
  end if
  return 0
end fn next_cell

fn step(g: List[List[Int]]) -> List[List[Int]]
  var nxt = empty_grid()
  for r in 0 until 8
    for c in 0 until 10
      nxt[r][c] = next_cell(alive: g[r][c], n: neighbours(g: g, r: r, c: c))
    end for
  end for
  return nxt
end fn step

fn count_live(g: List[List[Int]]) -> Int
  var n = 0
  for row in g
    for v in row
      n = n + v
    end for
  end for
  return n
end fn count_live

fn main() uses io
  var g = empty_grid()
  g[0][1] = 1
  g[1][2] = 1
  g[2][0] = 1
  g[2][1] = 1
  g[2][2] = 1
  g[5][9] = 1
  g[6][9] = 1
  g[7][9] = 1

  var line = ""
  for gen in 1 through 6
    g = step(g)
    let live = count_live(g)
    if line == ""
      line = "{live}"
    else
      line = "{line} {live}"
    end if
  end for
  print(line)

  for r in 0 until 8
    var row = ""
    for c in 0 until 10
      if g[r][c] == 1
        row = "{row}#"
      else
        row = "{row}."
      end if
    end for
    print(row)
  end for
end fn main
```

## adventure

Task: Write a small text adventure that runs a fixed list of commands. Rooms: Hall, Library, Kitchen, Vault. Exits: from Hall, north goes to Library and east to Kitchen; from Library, south goes to Hall; from Kitchen, west goes to Hall and down to Vault; from Vault, up goes to Kitchen. The way down from Kitchen is locked unless the player carries the key. Items at the start: Library has key, Kitchen has bread, Vault has gold. The player starts in Hall carrying nothing. Commands: `go <direction>` moves and prints the new room's name, or prints `You can't go <direction>.` if there's no such exit, or `The way down is locked.` if it's the locked way and the player has no key. `take <item>` moves the item from the room to the player and prints `Taken: <item>.`, or prints `There is no <item> here.`. `look` prints the room's name, then a line with the room's items separated by single spaces, or `nothing`. `inventory` prints the carried items in the order they were taken, separated by single spaces, or `nothing`. Run these commands in order: go east, take key, take bread, go down, go west, go north, look, take key, go south, go east, go down, take gold, look, go up, go up, inventory.

### attempt 1: OK

```cl
struct Item
  name: String
  place: String
end struct Item

struct World
  room: String
  items: List[Item]
  carried: List[String]
end struct World

# The room reached by going `dir` from `room`, or "" if there is no exit.
fn destination(room: String, dir: String) -> String
  example destination(room: "Hall", dir: "north") == "Library"
  example destination(room: "Vault", dir: "down") == ""
  if room == "Hall" and dir == "north"
    return "Library"
  else if room == "Hall" and dir == "east"
    return "Kitchen"
  else if room == "Library" and dir == "south"
    return "Hall"
  else if room == "Kitchen" and dir == "west"
    return "Hall"
  else if room == "Kitchen" and dir == "down"
    return "Vault"
  else if room == "Vault" and dir == "up"
    return "Kitchen"
  else
    return ""
  end if
end fn destination

# Joins words with single spaces.
fn joinWords(words: List[String]) -> String
  example joinWords(words: ["key", "bread"]) == "key bread"
  example joinWords(words: []) == ""
  var line = ""
  var first = true
  for word in words
    if first
      line = word
      first = false
    else
      line = "{line} {word}"
    end if
  end for
  return line
end fn joinWords

fn carrying(w: World, name: String) -> Bool
  for x in w.carried
    if x == name
      return true
    end if
  end for
  return false
end fn carrying

fn itemHere(w: World, name: String) -> Bool
  for it in w.items
    if it.name == name and it.place == w.room
      return true
    end if
  end for
  return false
end fn itemHere

fn goTo(w: World, dir: String) -> World uses io
  let dest = destination(room: w.room, dir: dir)
  if dest == ""
    print("You can't go {dir}.")
    return w
  end if
  if w.room == "Kitchen" and dir == "down" and not carrying(w: w, name: "key")
    print("The way down is locked.")
    return w
  end if
  print(dest)
  var r = w
  r.room = dest
  return r
end fn goTo

fn takeItem(w: World, name: String) -> World uses io
  if not itemHere(w: w, name: name)
    print("There is no {name} here.")
    return w
  end if
  var r = w
  for i in 0 until length(r.items)
    if r.items[i].name == name and r.items[i].place == r.room
      r.items[i].place = "player"
    end if
  end for
  r.carried = append(list: r.carried, item: name)
  print("Taken: {name}.")
  return r
end fn takeItem

fn doLook(w: World) uses io
  print(w.room)
  var here: List[String] = []
  for it in w.items
    if it.place == w.room
      here = append(list: here, item: it.name)
    end if
  end for
  if length(here) == 0
    print("nothing")
  else
    print(joinWords(words: here))
  end if
end fn doLook

fn doInventory(w: World) uses io
  if length(w.carried) == 0
    print("nothing")
  else
    print(joinWords(words: w.carried))
  end if
end fn doInventory

fn main() uses io
  var nothingYet: List[String] = []
  var w = World(room: "Hall", items: [Item(name: "key", place: "Library"), Item(name: "bread", place: "Kitchen"), Item(name: "gold", place: "Vault")], carried: nothingYet)
  w = goTo(w: w, dir: "east")
  w = takeItem(w: w, name: "key")
  w = takeItem(w: w, name: "bread")
  w = goTo(w: w, dir: "down")
  w = goTo(w: w, dir: "west")
  w = goTo(w: w, dir: "north")
  doLook(w: w)
  w = takeItem(w: w, name: "key")
  w = goTo(w: w, dir: "south")
  w = goTo(w: w, dir: "east")
  w = goTo(w: w, dir: "down")
  w = takeItem(w: w, name: "gold")
  doLook(w: w)
  w = goTo(w: w, dir: "up")
  w = goTo(w: w, dir: "up")
  doInventory(w: w)
end fn main
```
