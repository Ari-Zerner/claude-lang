## sieve

Task: Use the sieve of Eratosthenes (a list of Bools) to find the primes below 100. Print how many there are, then the largest one.

### attempt 1: OK

```cl
fn sieve(n: Int) -> List[Bool]
  example length(sieve(n: 10)) == 10
  var flags = repeat(item: true, times: n)
  var i = 0
  while i < n
    if i < 2
      flags[i] = false
    end if
    i = i + 1
  end while
  var p = 2
  while p * p < n
    if flags[p]
      var m = p * p
      while m < n
        flags[m] = false
        m = m + p
      end while
    end if
    p = p + 1
  end while
  return flags
end fn sieve

fn main() uses io
  let flags = sieve(n: 100)
  var count = 0
  var largest = 0
  var i = 0
  while i < length(flags)
    if flags[i]
      count = count + 1
      largest = i
    end if
    i = i + 1
  end while
  print(count)
  print(largest)
end fn main
```

## words

Task: Given the list of words apple, kiwi, banana, fig: print the longest word, then the total number of characters in all the words.

### attempt 1: OK

```cl
fn longest(words: List[String]) -> String
  example longest(["apple", "kiwi"]) == "apple"
  example longest(["fig", "kiwi"]) == "kiwi"
  var best = words[0]
  var i = 1
  while i < length(words)
    if length(words[i]) > length(best)
      best = words[i]
    end if
    i = i + 1
  end while
  return best
end fn longest

fn total_chars(words: List[String]) -> Int
  example total_chars(["ab", "c"]) == 3
  example total_chars([]) == 0
  var sum = 0
  var i = 0
  while i < length(words)
    sum = sum + length(words[i])
    i = i + 1
  end while
  return sum
end fn total_chars

fn main() uses io
  let words = ["apple", "kiwi", "banana", "fig"]
  print(longest(words))
  print(total_chars(words))
end fn main
```

## sort

Task: Sort [5, 2, 8, 1, 9, 3] with a sorting algorithm you write yourself and print the sorted list.

### attempt 1: OK

```cl
fn selectionSort(xs: List[Int]) -> List[Int]
  example selectionSort([5, 2, 8, 1, 9, 3]) == [1, 2, 3, 5, 8, 9]
  example selectionSort([]) == []
  var a = xs
  var i = 0
  while i < length(a)
    var m = i
    var j = i + 1
    while j < length(a)
      if a[j] < a[m]
        m = j
      end if
      j = j + 1
    end while
    var tmp = a[i]
    a[i] = a[m]
    a[m] = tmp
    i = i + 1
  end while
  return a
end fn selectionSort

fn main() uses io
  print(selectionSort([5, 2, 8, 1, 9, 3]))
end fn main
```
